## 문제의 내용
[문제 링크](https://www.acmicpc.net/problem/15682)

유효숫자가 최소 $10$개는 맞아야 하므로 정확한 계산을 요구하는 문제입니다. 
삼차 다항식이 정수 해를 항상 갖는다는 것이 문제 조건으로 주어져 있고 해의 범위도 제한되어 있습니다. 
한편 주어지는 방정식의 계수가 유한소수, 즉 유리수로 주어져 있으므로 적당한 정수를 곱하여 정수 계수를 갖는 다항식으로 전환할 수 있습니다. 
그리고 삼차 다항식이 정수 해를 가지면 $\mathbb{Z}[x]$의 원소로 변환된 다항식은 $\mathbb{Z}$ 상에서 분해됩니다. 

## 사용할 문제해결 전략
---
입력: 삼차 다항식 $f(x) = ax^3 + bx^2 + cx + d \in \polynomialring{\quotient}{x}$
1. 주어지는 계수를 `double`과 유리수(`python3`에서는 `fractions.Fraction`, C#에서는 자체 제작) 자료형으로 읽는다. 
2. 다항식 계수에 대해, [이곳](https://en.wikipedia.org/wiki/Cubic_equation#Discriminant_and_nature_of_the_roots)에서 제시된 판별식값(discriminant) $D$를 계산한다. 
3. $D$의 값에 따라 서로 다른 해를 구한다. 
    - $D = 0$이면 변곡점 $x = -\frac{b}{3a}$에서 삼차함수의 값을 유리수 계수를 이용해 계산하고, 
        - 값이 $0$ 미만이면 해는 $-\frac{b}{3a} - \frac{1}{3}\sqrt{b^2 - 3ac}$, $-\frac{b}{3a} + \frac{2}{3a}\sqrt{b^2 - 3ac}$이다. 
        - 값이 $0$ 이상이면 해는 $-\frac{b}{3a} - \frac{2}{3a}\sqrt{b^2 - 3ac}$, $-\frac{b}{3a} + \frac{1}{3a}\sqrt{b^2 - 3ac}$이다. 
        - 값이 $0$인 경우는 $-\frac{b}{3a}$가 유일한 실수 해이고 정수이다. 
    - $D < 0$이면 $-1,000,000$ 이상 $1,000,000$ 이하의 실직선 상에서 이분 탐색을 실행한다. 
    - $D > 0$이면, `double`로 저장된 다항식 계수를 사용하면서 다음 세 구간 사이에 있는 정수들에 대하여 이분탐색을 실행한다. 
        
        - $\intopenopen{-\frac{b}{3a} - \frac{2}{3a}\sqrt{b^2 - 3ac}}{-\frac{b}{3a} - \frac{1}{3}\sqrt{b^2 - 3ac}}$
        - $\intopenopen{-\frac{b}{3a} - \frac{1}{3a}\sqrt{b^2 - 3ac}}{-\frac{b}{3a} + \frac{1}{3}\sqrt{b^2 - 3ac}}$
        - $\intopenopen{-\frac{b}{3a} + \frac{1}{3a}\sqrt{b^2 - 3ac}}{-\frac{b}{3a} + \frac{2}{3}\sqrt{b^2 - 3ac}}$
        
        이 중 가장 가까운 정수와의 거리가 가장 가까운 것을 정수 해로 간주한다.
        정수 해를 사용해 유리수 계수에 조립제법(synthetic method)을 적용하고, 남은 이차 인수에 이차방정식 근의 공식을 적용한다. 
6. 남은 다항식의 차수가: 
    - 이차식일 때는 이차방정식 근의 공식을 사용한다. 
    - 일차식일 때는 자명하게 해를 구한다. 
7. 해를 정렬하고 형식에 맞게 출력한다. 
---

## 제출한 소스
```csharp {tabWidth=2}
// https://www.acmicpc.net/problem/15682

using System.Numerics;

static Fraction EvaluateDiscriminant(Fraction[] coefficients) {
	ref var a = ref coefficients[3];
	ref var b = ref coefficients[2];
	ref var c = ref coefficients[1];
	ref var d = ref coefficients[0];
	return 18 * a * b * c * d
		- 4 * b * b * b * d
		+ b * b * c * c
		- 4 * a * c * c * c
		- 27 * a * a * d * d;
}
static double EvaluatePolynomial(double[] coefficient, double x) {
	double result = 0.0;
	foreach (var c in coefficient.Reverse()) {
		result = Math.FusedMultiplyAdd(x, result, c);
	}
	return result;
}
static double EvaluatePolynomialExact(Fraction[] coefficients, Fraction x) {
	Fraction result = new();
	foreach (var c in coefficients.Reverse()) {
		result = result * x + c;
	}
	return (double)result;
}
static double BinarySearch(double[] coefficient, double begin, double end, bool increasing = true) {
	double lo = begin, hi = end;
	while (Math.Abs(lo - hi) >= 1e-10) {
		var mid = (hi + lo) / 2;
		var midEval = EvaluatePolynomial(coefficient, mid);
		if (midEval > 0.0) {
			if (increasing) hi = mid;
			else lo = mid;
		}
		else if (midEval < 0.0) {
			if (increasing) lo = mid;
			else hi = mid;
		}
		else return mid;
	}
	return (lo + hi) / 2.0;
}

static double GetDiscriminantNegativeRoots(double[] coefficients) {
	return Math.Round(BinarySearch(coefficients, -1_000_000.0, 1_000_000.0));
}
static double[] GetDiscriminantZeroRoots(double[] coefficients, Fraction[] fractionCoefficients) {
	var a = coefficients[3];
	var b = coefficients[2];
	var c = coefficients[1];
	var d = coefficients[0];
	double turningPoint = -b / a / 3.0;
	var turningPointEval = EvaluatePolynomialExact(fractionCoefficients, -fractionCoefficients[2] / fractionCoefficients[3] / 3);
	if (turningPointEval != 0.0) {
		double baseWidth = 1.0 / 3.0 * Math.Sqrt(
			(double)(fractionCoefficients[2] * fractionCoefficients[2] - 3 * fractionCoefficients[3] * fractionCoefficients[1])
		) / (double)fractionCoefficients[3];
		if (turningPointEval > 0.0) return new double[] { turningPoint - 2.0 * baseWidth, turningPoint + baseWidth };
		else return new double[] { turningPoint - baseWidth, turningPoint + 2.0 * baseWidth };
	}
	else return new double[] { Math.Round(turningPoint) };
}
static double[] GetDiscriminantPositiveRoots(double[] coefficients, Fraction[] fractionCoefficients) {
	ref var a = ref coefficients[3];
	ref var b = ref coefficients[2];
	ref var c = ref coefficients[1];
	ref var d = ref coefficients[0];
	double baseWidth = 1.0 * Math.Sqrt((double)(b * b - 3 * a * c)) / 3.0;
	double turningPoint = -b / a / 3.0;
	var root_1 = BinarySearch(coefficients, turningPoint - 2 * baseWidth, turningPoint - baseWidth);
	var root_2 = BinarySearch(coefficients, turningPoint - baseWidth, turningPoint + baseWidth, false);
	var root_3 = BinarySearch(coefficients, turningPoint + baseWidth, turningPoint + 2 * baseWidth);
	int integerRoot = (int)Math.Round((new double[] { root_1, root_2, root_3 }).MinBy(x => Math.Abs(Math.Round(x) - x)));
	// synthetic method
	var quadraticFactor = new Fraction[3];
	quadraticFactor[2] = fractionCoefficients[3]; // 1
	quadraticFactor[1] = integerRoot * quadraticFactor[2] + fractionCoefficients[2];
	quadraticFactor[0] = integerRoot * quadraticFactor[1] + fractionCoefficients[1];
	var discriminant = (double)(quadraticFactor[1] * quadraticFactor[1] - 4 * quadraticFactor[2] * quadraticFactor[0]);
	var sqrtDiscriminant = Math.Sqrt(discriminant);
	var p = (double)quadraticFactor[2];
	var q = (double)quadraticFactor[1];
	var r = (double)quadraticFactor[0];
	// use numerically stable method 
	if (q >= 0) return new double[] { integerRoot, -2 * r / (q + sqrtDiscriminant), (-q - sqrtDiscriminant) / 2 / p };
	else return new double[] { integerRoot, 2 * r / (-q + sqrtDiscriminant), (-q + sqrtDiscriminant) / 2 / p };
}

int testCount = Convert.ToInt32(Console.ReadLine());
while (testCount-- > 0) {
	string[] stringCoefficients = Console.ReadLine()!.Split();
	Array.Reverse(stringCoefficients);
	Fraction[] coefficients = stringCoefficients.Select(x => Fraction.Parse(x)).ToArray();
	double[] doubleCoefficients = stringCoefficients.Select(x => double.Parse(x)).ToArray();
	coefficients = coefficients.Select(x => x / coefficients[^1]).ToArray();
	doubleCoefficients = doubleCoefficients.Select(x => x / doubleCoefficients[^1]).ToArray();
	var discriminant = EvaluateDiscriminant(coefficients);
	double[] roots;
	if (discriminant.IsZero) roots = GetDiscriminantZeroRoots(doubleCoefficients, coefficients);
	else if (discriminant.IsPositive) roots = GetDiscriminantPositiveRoots(doubleCoefficients, coefficients);
	else roots = new double[] { GetDiscriminantNegativeRoots(doubleCoefficients) };
	Array.Sort(roots);
	roots = roots.Select(x => Math.Abs(x) < 1e-10 ? 0.0 : x).ToArray();
	Console.WriteLine(string.Join(" ", roots.Select(x => x.ToString("F10"))));
}
class Fraction {
	public BigInteger Numerator { get; private set; }
	public BigInteger Denominator { get; private set; }
	public bool IsZero => Numerator.IsZero;
	public bool IsPositive => Numerator.Sign > 0;
	public bool IsNegative => Numerator.Sign < 0;

	public Fraction() {
		Numerator = BigInteger.Zero;
		Denominator = BigInteger.One;
	}
	public Fraction(BigInteger num) {
		Numerator = new(num.ToByteArray());
		Denominator = BigInteger.One;
	}
	public Fraction(BigInteger num, BigInteger den) {
		if (den == BigInteger.Zero) throw new DivideByZeroException();
		Numerator = new(num.ToByteArray());
		Denominator = new(den.ToByteArray());
		Reduce();
	}
	private void Reduce() {
		if (Numerator.IsZero) {
			Denominator = 1;
			return;
		}
		var gcd = BigInteger.GreatestCommonDivisor(BigInteger.Abs(Numerator), Denominator);
		Numerator /= gcd;
		Denominator /= gcd;
	}
	public static Fraction Parse(string x) {
		var split = x.Split('.');
		if (split.Length == 1) return new(BigInteger.Parse(split[0]));
		else return new(BigInteger.Parse(split[0] + split[1]), BigInteger.Parse("1" + new string('0', split[1].Length)));
	}
	public static Fraction operator +(Fraction a, Fraction b) {
		return new(a.Numerator * b.Denominator + a.Denominator * b.Numerator, a.Denominator * b.Denominator);
	}
	public static Fraction operator -(Fraction a, Fraction b) {
		return new(a.Numerator * b.Denominator - a.Denominator * b.Numerator, a.Denominator * b.Denominator);
	}
	public static Fraction operator -(Fraction x) {
		return new(-x.Numerator, x.Denominator);
	}
	public static Fraction operator *(Fraction a, Fraction b) {
		return new(a.Numerator * b.Numerator, a.Denominator * b.Denominator);
	}
	public static Fraction operator *(Fraction a, BigInteger b) {
		return new(a.Numerator * b, a.Denominator);
	}
	public static Fraction operator *(BigInteger a, Fraction b) {
		return new(a * b.Numerator, b.Denominator);
	}
	public static Fraction operator /(Fraction a, Fraction b) {
		if (b.IsZero) throw new DivideByZeroException();
		if (b.IsNegative) return new(-a.Numerator * b.Denominator, a.Denominator * -b.Numerator);
		else return new(a.Numerator * b.Denominator, a.Denominator * b.Numerator);
	}
	public static Fraction operator /(Fraction a, BigInteger b) {
		if (b.IsZero) throw new DivideByZeroException();
		if (b.Sign < 0) return new(-a.Numerator, a.Denominator * -b);
		else return new(a.Numerator, a.Denominator * b);
	}
	public static Fraction operator /(BigInteger a, Fraction b) {
		if (b.IsZero) throw new DivideByZeroException();
		return new(a * b.Numerator, b.Denominator);
	}
	public static explicit operator double(Fraction x) {
		if (x.IsZero) return 0.0;
		var quotient = BigInteger.DivRem(x.Numerator, x.Denominator, out var remainder);
		return (double)quotient + (double)remainder / (double)x.Denominator;
	}
}
```

## 특이사항
변곡점에서 부호의 정확한 계산을 위해서 부동소수점 연산이 아닌 유리수에 의한 정확한 계산이 사용되었습니다. 
초반 시도에서는 모든 단계에서 정확한 계산을 위해 유리수 연산을 전반적으로 사용했으나, 성능 상의 문제로 다음 상황에서 유리수 연산이 사용되었습니다. 
- 삼차다항식 판별식의 정확한 부호 판정을 위한 연산
- 변곡점에서의 함숫값의 정확한 부호 판정을 위한 연산
- 서로 다른 두 개의 실수 해를 구할 때 $b^2 - 3ac$의 정확한 계산
- 서로 다른 세 개의 실수 해를 구할 때 이차인수의 판별식의 정확한 계산
그 외에는 근사치를 구하거나 최종적인 답안을 제출하는 용도로 배정밀도(`double`) 실수자료형을 사용하였습니다. 

판별식을 정확하게 계산해야 하는 상황은 처음에 인지하지 못 했습니다. 
먼저 다음 다항식을 살펴봅시다:

\\[
  \begin{aligned}
  &(x - 1)(x - 999.99999)(x - 1000.00001) \\\\
  = &(x - 1)(x^2 - 2000x + 999999.9999999999)
  \end{aligned}
\\]

정수 해 $1$을 구하고 나면 위의 이차인수의 해를 계산해야 합니다. 
근의 공식을 사용하기 위해 판별식을 계산하면 $0.0000000004 = 4 \times 10^{-10}$임을 확인할 수 있습니다. 
그런데 이를 부동소수점을 통해 계산하게 된다면 유효숫자 상실이 발생합니다. 
$2000 \times 2000 = 4000000$, $4 \times 999999.9999999999 = 3999999.9999999996$인데, 
배정밀도의 십진 유효숫자는 $15.95$ 정도로 할 수 있고, 배정밀도에서는 
$$3999999.9999999996 = 4000000.000000000$$
입니다. 
따라서 판별식이 $0$에 매우 근접하게 되어 제곱근의 계산에도 영향을 주게 되고, 판별식 제곱근의 참값인 $0.00002$와 괴리된 값을 얻을 수 있습니다. 
이에 판별식의 유효숫자를 모두 보존하기 위해 유리수 상에서 먼저 계산 후 부동소수점 자료형으로 형변환을 시도하는 것입니다. 

그 외에는 수학적인 조건에 따라 경우를 나누는 것으로 결정하는 문제였습니다.
